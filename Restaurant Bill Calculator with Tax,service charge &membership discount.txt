food_amount = float(input())
day = input()
membership = input()
# GST = 5%
gst = food_amount * 0.05
amount = food_amount + gst
# Service charge
if day == "Saturday" or day == "Sunday":
    service_charge = food_amount * 0.10
else:
    service_charge = food_amount * 0.05
amount = amount + service_charge
# Membership discount
if membership == "Gold":
    discount = amount * 0.08
elif membership == "Silver":
    discount = amount * 0.04
else:
    discount = 0
final_bill = amount - discount
# Cashback
if final_bill > 3000:
    cashback = final_bill * 0.02
else:
    cashback = 0
print(f"{final_bill:.2f}")
print(f"{cashback:.2f}")
